/* W7DIAG: unified diagnostic flight recorder (W7). Entire file is W7-only. */
/* W7DIAG: RAM ring in .uninitialized_data (NOLOAD: survives WDT reboot). */
/* W7DIAG: Single-writer discipline: Core0 context only. No IRQ writes, */
/* W7DIAG: no printf, no flash writes, no mutex. Deleted at W7 restore. */
#pragma once /* W7DIAG */
#include <stdint.h> /* W7DIAG */
#include <stdbool.h> /* W7DIAG */
#include <stddef.h> /* W7DIAG */
#ifdef __cplusplus /* W7DIAG */
extern "C" { /* W7DIAG */
#endif /* W7DIAG */
/* W7DIAG: recorder magic 0xD1A6nnnn (nnnn = format id 0007). */
/* W7DIAG */ #define W7_MAGIC 0xD1A60007u
/* W7DIAG */ #define W7_VER 7u
/* W7DIAG */ #define W7_NRING 96u
/* W7DIAG: watchdog scratch ownership: we use [0..2]; [3] spare unused. */
/* W7DIAG: SDK owns [4..7] (watchdog_enable/reboot magic + pc/sp). Never touch. */
/* W7DIAG */ #define W7_SCR_MAGIC 0
/* W7DIAG */ #define W7_SCR_INDEX 1
/* W7DIAG */ #define W7_SCR_STAGE 2
/* W7DIAG: entry kinds. */
/* W7DIAG */ #define W7K_FLASH 1u
/* W7DIAG */ #define W7K_STAGE 2u
/* W7DIAG */ #define W7K_TIMER 3u
/* W7DIAG */ #define W7K_EV 4u
/* W7DIAG: flash-op ids (FLASH entry a1 = (op<<1)|phase). */
/* W7DIAG */ #define W7OP_STORE_SAFE 1u
/* W7DIAG */ #define W7OP_STORE_TAG 2u
/* W7DIAG */ #define W7OP_DELETE_TAG 3u
/* W7DIAG */ #define W7OP_LK_PUT 4u
/* W7DIAG */ #define W7OP_LK_DEL 5u
/* W7DIAG */ #define W7PH_BEGIN 0u
/* W7DIAG */ #define W7PH_END 1u
/* W7DIAG: owned-timer ids (usb/empty/stats only; reconnect_timer untouched). */
/* W7DIAG */ #define W7TMR_USB 1u
/* W7DIAG */ #define W7TMR_EMPTY 2u
/* W7DIAG */ #define W7TMR_STATS 3u
/* W7DIAG */ #define W7TA_ADD 1u
/* W7DIAG */ #define W7TA_REMOVE 2u
/* W7DIAG */ #define W7TA_FIRE 3u
/* W7DIAG: poll-vs-rearm stage chain (latest mirrored to scratch[2]). */
/* W7DIAG */ #define W7ST_POLL_ENTER 1u
/* W7DIAG */ #define W7ST_POLL_BODY_DONE 2u
/* W7DIAG */ #define W7ST_WDT_UPDATE_DONE 3u
/* W7DIAG */ #define W7ST_POLL_TAIL_DONE 4u
/* W7DIAG */ #define W7ST_POLL_RETURN_IMMINENT 5u
/* W7DIAG */ #define W7ST_USB_HANDLER_ENTER 6u
/* W7DIAG */ #define W7ST_POLL_RETURNED 7u
/* W7DIAG */ #define W7ST_SET_TIMER_DONE 8u
/* W7DIAG */ #define W7ST_ADD_TIMER_DONE 9u
/* W7DIAG */ #define W7ST_USB_HANDLER_RETURN_IMMINENT 10u
/* W7DIAG: one ring entry: timestamp_us + kind + args (16 B, fixed). */
/* W7DIAG */ typedef struct {
/* W7DIAG */     uint32_t ts;
/* W7DIAG */     uint16_t kind;
/* W7DIAG */     uint16_t a1;
/* W7DIAG */     uint32_t a2;
/* W7DIAG */     uint32_t a3;
/* W7DIAG */ } w7_entry_t;
/* W7DIAG: persisted state (lives in .uninitialized_data; CRC-guarded). */
/* W7DIAG: No secrets ever stored (no BD_ADDR, no keys, no tags payload). */
/* W7DIAG */ typedef struct {
/* W7DIAG */     uint32_t magic;
/* W7DIAG */     uint32_t crc;
/* W7DIAG */     uint32_t ver;
/* W7DIAG */     uint32_t idx;
/* W7DIAG */     uint32_t n;
/* W7DIAG */     uint32_t begins[6];
/* W7DIAG */     uint32_t ends[6];
/* W7DIAG */     uint32_t last_tag[6];
/* W7DIAG */     int32_t last_rc[6];
/* W7DIAG */     uint32_t last_dur[6];
/* W7DIAG */     uint32_t tmr_add[4];
/* W7DIAG */     uint32_t tmr_rem[4];
/* W7DIAG */     uint32_t tmr_fire[4];
/* W7DIAG */     uint32_t tmr_to[4];
/* W7DIAG */     uint32_t stage_last;
/* W7DIAG */     uint32_t stage_n;
/* W7DIAG */     uint8_t stage_trail[8];
/* W7DIAG */     uint32_t ev_last;
/* W7DIAG */     uint32_t ev_n;
/* W7DIAG */     w7_entry_t ring[W7_NRING];
/* W7DIAG */ } w7_state_t;
/* W7DIAG: init (keeps a valid recorder across WDT reboot; resets if bad). */
/* W7DIAG */ void diag_rec_init(void);
/* W7DIAG: record helpers (Core0 only; RAM-only; never print). */
/* W7DIAG */ void diag_rec_stage(uint8_t st);
/* W7DIAG: FLASH record: begin => a2 = tag (0 when n/a), a3 = len-or-0. */
/* W7DIAG: FLASH record: end   => a2 = rc (as u32), a3 = duration_us. */
/* W7DIAG */ void diag_rec_flash(uint8_t op, uint8_t phase, uint32_t a2, uint32_t a3);
/* W7DIAG */ void diag_rec_timer(uint8_t id, uint8_t act, uint32_t timeout_ms);
/* W7DIAG */ void diag_rec_ev(uint8_t ev);
/* W7DIAG: boot-dump support (boot context only; the recorder never prints). */
/* W7DIAG */ bool diag_rec_valid(void);
/* W7DIAG */ const w7_state_t *diag_rec_state(void);
#ifdef __cplusplus /* W7DIAG */
} /* W7DIAG */
#endif /* W7DIAG */
