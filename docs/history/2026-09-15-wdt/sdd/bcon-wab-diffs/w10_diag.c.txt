#define W10_MAGIC 0x57443130u /* W10DIAG */
#define W10_VER   1u /* W10DIAG */
#define W10_NRING 32u /* W10DIAG */

enum { /* W10DIAG */
    W10K_INIT = 1u, W10K_LK_REQ, W10K_LK_ACK, W10K_LK_END, /* W10DIAG */
    W10K_FSE_OK, W10K_FSE_FAIL, /* W10DIAG */
    W10K_ROMP_BEG, W10K_ROMP_END, W10K_ROME_BEG, W10K_ROME_END, /* W10DIAG */
    W10K_FORCE /* W10DIAG */
}; /* W10DIAG */

typedef struct { uint32_t ts, epoch, kind, a, b; } w10_ev_t; /* W10DIAG */

typedef struct { /* W10DIAG */
    uint32_t magic, ver, crc; /* W10DIAG */
    uint32_t post_epoch, boot_epoch; /* W10DIAG */
    uint32_t idx, n; /* W10DIAG */
    w10_ev_t ring[W10_NRING]; /* W10DIAG */
    uint32_t lk_req, lk_ack, lk_fail, lk_end; /* W10DIAG */
    uint32_t fse_ok, fse_fail, fse_cap; /* W10DIAG */
    uint32_t romp_beg, romp_end, rome_beg, rome_end; /* W10DIAG */
    uint32_t force_gen, force_n; /* W10DIAG */
    int32_t  last_rc; /* W10DIAG */
    uint32_t last_dur; /* W10DIAG */
} w10_state_t; /* W10DIAG */

static w10_state_t w10_s __attribute__((section(".uninitialized_data"))); /* W10DIAG */

/* W10DIAG: FNV-1a/32 over the whole state with the crc word forced to zero. */ /* W10DIAG */
static uint32_t w10_crc(void) { /* W10DIAG */
    const uint8_t *p = (const uint8_t *)&w10_s; /* W10DIAG */
    size_t i; /* W10DIAG */
    uint32_t h = 2166136261u; /* W10DIAG */
    size_t co = (size_t)((const uint8_t *)&w10_s.crc - p); /* W10DIAG */
    for (i = 0; i < sizeof(w10_s); i++) { /* W10DIAG */
        uint8_t b = (i >= co && i < co + sizeof(uint32_t)) ? 0u : p[i]; /* W10DIAG */
        h = (h ^ b) * 16777619u; /* W10DIAG */
    } /* W10DIAG */
    return h; /* W10DIAG */
} /* W10DIAG */

void w10_ev(uint32_t kind, uint32_t a, uint32_t b) { /* W10DIAG */
    w10_ev_t *e = &w10_s.ring[w10_s.idx % W10_NRING]; /* W10DIAG */
    e->ts = time_us_32(); /* W10DIAG */
    e->epoch = w10_s.boot_epoch; /* W10DIAG */
    e->kind = kind; /* W10DIAG */
    e->a = a; /* W10DIAG */
    e->b = b; /* W10DIAG */
    w10_s.idx++; /* W10DIAG */
    w10_s.n++; /* W10DIAG */
    w10_s.crc = w10_crc(); /* W10DIAG */
} /* W10DIAG */

void w10_init(void) { /* W10DIAG */
    bool ok = (w10_s.magic == W10_MAGIC) && (w10_s.ver == W10_VER) && /* W10DIAG */
              (w10_s.crc == w10_crc()); /* W10DIAG */
    if (ok) { /* W10DIAG */
        w10_s.post_epoch = w10_s.boot_epoch; /* W10DIAG */
    } else { /* W10DIAG */
        memset(&w10_s, 0, sizeof(w10_s)); /* W10DIAG */
        w10_s.magic = W10_MAGIC; /* W10DIAG */
        w10_s.ver = W10_VER; /* W10DIAG */
        w10_s.post_epoch = 0u; /* W10DIAG */
    } /* W10DIAG */
    w10_s.boot_epoch = w10_s.post_epoch + 1u; /* W10DIAG */
    w10_ev(W10K_INIT, 0u, 0u); /* W10DIAG */
} /* W10DIAG */

uint32_t w10_force_fire(void) { /* W10DIAG */
    w10_s.force_gen++; /* W10DIAG */
    w10_s.force_n++; /* W10DIAG */
    w10_ev(W10K_FORCE, w10_s.force_gen, w10_s.force_n); /* W10DIAG */
    return w10_s.force_gen; /* W10DIAG */
} /* W10DIAG */

/* W10DIAG: P2/P4 point A: flash_safe_execute wrapper. W10_TIMEOUT_MS>0 caps the HAL's */ /* W10DIAG */
/* W10DIAG: UINT32_MAX timeout (diagnostic only; NOT a permanent fix). rc is recorded. */ /* W10DIAG */
int __wrap_flash_safe_execute(void (*func)(void *), void *param, uint32_t timeout_ms) { /* W10DIAG */
    uint32_t eff = timeout_ms; /* W10DIAG */
#if W10_TIMEOUT_MS > 0 /* W10DIAG */
    if (timeout_ms == UINT32_MAX) { /* W10DIAG */
        eff = (uint32_t)W10_TIMEOUT_MS; /* W10DIAG */
        w10_s.fse_cap++; /* W10DIAG */
    } /* W10DIAG */
#endif /* W10DIAG */
    uint32_t t0 = time_us_32(); /* W10DIAG */
    int rc = __real_flash_safe_execute(func, param, eff); /* W10DIAG */
    uint32_t dur = time_us_32() - t0; /* W10DIAG */
    w10_s.last_rc = (int32_t)rc; /* W10DIAG */
    w10_s.last_dur = dur; /* W10DIAG */
    if (rc == PICO_OK) { /* W10DIAG */
        w10_s.fse_ok++; /* W10DIAG */
        w10_ev(W10K_FSE_OK, (uint32_t)rc, dur); /* W10DIAG */
    } else { /* W10DIAG */
        w10_s.fse_fail++; /* W10DIAG */
        w10_ev(W10K_FSE_FAIL, (uint32_t)rc, dur); /* W10DIAG */
    } /* W10DIAG */
    return rc; /* W10DIAG */
} /* W10DIAG */

/* W10DIAG: P1/P2: lockout request just before, ack just after (requester side). */ /* W10DIAG */
bool __wrap_multicore_lockout_start_timeout_us(uint64_t timeout_us) { /* W10DIAG */
    uint32_t inf = (timeout_us >= 1000000000ull) ? 1u : 0u; /* W10DIAG */
    uint32_t ms = inf ? 0u : (uint32_t)(timeout_us / 1000ull); /* W10DIAG */
    w10_s.lk_req++; /* W10DIAG */
    w10_ev(W10K_LK_REQ, inf, ms); /* W10DIAG */
    uint32_t t0 = time_us_32(); /* W10DIAG */
    bool ok = __real_multicore_lockout_start_timeout_us(timeout_us); /* W10DIAG */
    uint32_t dur = time_us_32() - t0; /* W10DIAG */
    if (ok) w10_s.lk_ack++; else w10_s.lk_fail++; /* W10DIAG */
    w10_ev(W10K_LK_ACK, ok ? 1u : 0u, dur); /* W10DIAG */
    return ok; /* W10DIAG */
} /* W10DIAG */

bool __wrap_multicore_lockout_end_timeout_us(uint64_t timeout_us) { /* W10DIAG */
    uint32_t t0 = time_us_32(); /* W10DIAG */
    bool ok = __real_multicore_lockout_end_timeout_us(timeout_us); /* W10DIAG */
    w10_s.lk_end++; /* W10DIAG */
    w10_ev(W10K_LK_END, ok ? 1u : 0u, time_us_32() - t0); /* W10DIAG */
    return ok; /* W10DIAG */
} /* W10DIAG */

/* W10DIAG: P3/P4: ROM (QMI program/erase) call just before / just after. */ /* W10DIAG */
void __wrap_flash_range_program(uint32_t flash_offs, const uint8_t *data, size_t count) { /* W10DIAG */
    w10_s.romp_beg++; /* W10DIAG */
    w10_ev(W10K_ROMP_BEG, flash_offs, (uint32_t)count); /* W10DIAG */
    uint32_t t0 = time_us_32(); /* W10DIAG */
    __real_flash_range_program(flash_offs, data, count); /* W10DIAG */
    w10_s.romp_end++; /* W10DIAG */
    w10_ev(W10K_ROMP_END, flash_offs, time_us_32() - t0); /* W10DIAG */
} /* W10DIAG */

void __wrap_flash_range_erase(uint32_t flash_offs, size_t count) { /* W10DIAG */

    w10_s.rome_end++; /* W10DIAG */
    w10_ev(W10K_ROME_END, flash_offs, time_us_32() - t0); /* W10DIAG */
} /* W10DIAG */
